```asm

Str_nextWord PROC USES eax ecx edx esi,
    pString:PTR BYTE,
    delimiter:DWORD

    mov esi, pString
    mov dl, BYTE PTR delimiter

search_loop:
    mov al, [esi]
    cmp al, 0
    je not_found

    cmp al, dl
    je found

    inc esi
    jmp search_loop

found:
    mov BYTE PTR [esi], 0
    inc esi
    mov eax, esi
    ret

not_found:
    or eax, 1
    ret

Str_nextWord ENDP
