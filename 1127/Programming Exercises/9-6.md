```asm
Get_frequencies PROC USES eax ebx ecx esi edi,
    pString:PTR BYTE,
    pTable:PTR DWORD

    mov esi, pString
    mov edi, pTable

NextChar:
    mov al, [esi]
    cmp al, 0
    je  Done

    movzx ebx, al
    mov ecx, [edi + ebx*4]
    inc ecx
    mov [edi + ebx*4], ecx

    inc esi
    jmp NextChar

Done:
    ret
Get_frequencies ENDP
