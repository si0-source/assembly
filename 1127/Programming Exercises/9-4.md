```asm
Str_find PROC USES eax ebx ecx edx esi edi,pSource:PTR BYTE,pTarget:PTR BYTE

    mov esi, pTarget
outer_loop:
    mov al, [esi]
    cmp al, 0
    je not_found

    mov edi, pSource
    mov ebx, esi

inner_loop:
    mov al, [ebx]
    mov dl, [edi]
    cmp dl, 0
    je found

    cmp al, dl
    jne no_match

    inc ebx
    inc edi
    jmp inner_loop

no_match:
    inc esi
    jmp outer_loop

found:
    mov eax, esi
    ret

not_found:
    or eax, 1
    ret
Str_find ENDP
