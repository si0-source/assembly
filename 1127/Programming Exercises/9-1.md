'''asm
Str_copyN PROC USES eax ecx esi edi,
    source:PTR BYTE,
    target:PTR BYTE,
    maxCount:DWORD

    INVOKE Str_length, source
    inc eax

    mov ecx, eax
    mov edx, maxCount

    cmp ecx, edx
    jbe size_ok
    mov ecx, edx

size_ok:
    mov esi, source
    mov edi, target

    cld
    rep movsb

    ret
Str_copyN ENDP
