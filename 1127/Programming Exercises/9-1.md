Str_copyN PROC USES eax ecx esi edi,
    source:PTR BYTE,     ; source string
    target:PTR BYTE,     ; target string
    maxCount:DWORD       ; maximum characters to copy

    INVOKE Str_length, source     ; EAX = length(src)
    inc eax                       ; include null byte

    mov ecx, eax                  ; ECX = source length + null
    mov edx, maxCount             ; limit

    cmp ecx, edx
    jbe size_ok
    mov ecx, edx                  ; ECX = min(length+1, maxCount)

size_ok:
    mov esi, source
    mov edi, target

    cld
    rep movsb                     ; copy ECX bytes max

    ret
Str_copyN ENDP
